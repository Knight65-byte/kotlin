fun main() {
    println("--- QUESTÃO 1 ---")
    println("Com PROMO10: R$ ${calcularDesconto(100.0, "PROMO10")}")
    println("Com PROMO20: R$ ${calcularDesconto(100.0, "PROMO20")}")
    println("Com Cupom Inválido: R$ ${calcularDesconto(100.0, "INVALIDO")}")
    println("Sem Cupom (null): R$ ${calcularDesconto(100.0, null)}")

    println("\n--- QUESTÃO 2 ---")
    val enderecos = listOf("Rua das Flores, 123", null, "Av. Brasil, 456", null, "Rua XV de Novembro, 789")
    auditarEntregas(enderecos)

    println("\n--- QUESTÃO 3 ---")
    validarBioInfantil("Gosto de desenhar e brincar no parque.")
    validarBioInfantil("Esta é uma biografia absurdamente longa para uma criança pequena que ultrapassa cinquenta caracteres de limite com facilidade.")
    validarBioInfantil(null)

    println("\n--- QUESTÃO 4 ---")
    processarTransacoesPix()

    println("\n--- QUESTÃO 5 ---")
    avaliarMotorista(5)
    avaliarMotorista(3)
    avaliarMotorista(null)

    println("\n--- QUESTÃO 6 ---")
    println("Gorjeta de R$ 15.0: R$ ${calcularGorjeta(15.0)}")
    println("Gorjeta nula: R$ ${calcularGorjeta(null)}")
    println("Gorjeta negativa: R$ ${calcularGorjeta(-5.0)}")

    println("\n--- QUESTÃO 7 ---")
    val listaEmails = listOf("usuario@email.com", null, "", "   ", "dev@kotlin.org")
    limparBancoDeDados(listaEmails)

// ==========================================
// QUESTÃO 1: Sistema de Cupons Avançado
// ==========================================
fun calcularDesconto(valor: Double, cupom: String?): Double = when (cupom) {
    "PROMO10" -> valor - 10.0
    "PROMO20" -> valor - 20.0
    else -> valor
}

// ==========================================
// QUESTÃO 2: Auditoria de Entregas
// ==========================================
fun auditarEntregas(enderecos: List<String?>) {
    for (item in enderecos) {
        val enderecoFinal = item ?: "Endereço Desconhecido"
        if (enderecoFinal == "Endereço Desconhecido") {
            println("Entrega Pendente: Falta de dados")
        } else {
            println("Rota traçada para: $enderecoFinal")
        }
    }
}

// ==========================================
// QUESTÃO 3: Validação de Perfil de Streaming
// ==========================================
fun validarBioInfantil(bio: String?) {
    val tamanho = bio?.length ?: 0
    if (tamanho <= 50) {
        println("Bio aceita")
    } else {
        println("Bio muito longa")
    }
}

// ==========================================
// QUESTÃO 4: Processamento de Transações Pix
// ==========================================
fun processarTransacoesPix() {
    val transacoes: List<Double?> = listOf(50.0, null, 120.5, null, 10.0)

    val total = transacoes.sumOf { valor ->
        if (valor != null) {
            valor
        } else {
            println("Transação ignorada")
            0.0
        }
    }

    println("Total processado: R$ $total")
}

// ==========================================
// QUESTÃO 5: Classificação de Feedback de Motoristas
// ==========================================
fun avaliarMotorista(nota: Int?) {
    when (nota ?: 0) {
        5 -> println("Excelente corrida!")
        4 -> println("Boa corrida.")
        1, 2, 3 -> println("Precisamos melhorar.")
        0 -> println("Nenhuma avaliação fornecida.")
        else -> println("Nota fora do intervalo válido.")
    }
}

// ==========================================
// QUESTÃO 6: Função Lambda para Cálculo de Gorjeta
// ==========================================
val calcularGorjeta: (Double?) -> Double = {
    if ((it ?: 0.0) > 0.0) it!! else 0.0
}

// ==========================================
// QUESTÃO 7: Limpeza de Banco de Dados de Usuários
// ==========================================
fun limparBancoDeDados(emails: List<String?>) {
    var contasInvalidas = 0

    for (email in emails) {
        // Safe call com elvis operator tratando verificação de string em branco/nula
        val ehInvalido = email?.isBlank() ?: true

        if (ehInvalido) {
            contasInvalidas++
            println("Aviso de deleção: Conta inválida e marcada para remoção.")
        } else {
            println("Conta válida: $email")
        }
    }

    println("Total de contas que precisam ser apagadas: $contasInvalidas")
}
